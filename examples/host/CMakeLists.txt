cmake_minimum_required(VERSION 3.16)
project(cambold_host_example LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(_default_root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CAMBOLD_SOURCE_DIR "${_default_root}" CACHE PATH "CamBold project root")
if(NOT EXISTS "${CAMBOLD_SOURCE_DIR}/cambold_sdk/include/cb_sdk.h")
  message(FATAL_ERROR "Set CAMBOLD_SOURCE_DIR to the CamBold project root")
endif()
set(CB_SDK_BACKEND USB CACHE STRING "")
add_subdirectory("${CAMBOLD_SOURCE_DIR}/cambold_sdk" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
add_executable(cambold_example main.cpp)
target_link_libraries(cambold_example PRIVATE CamBold::SDK)
if(MINGW)
  target_link_options(cambold_example PRIVATE -static -static-libgcc -static-libstdc++)
endif()
